```yaml
rules:
  - id: external-call-state-mutation
    patterns:
      - pattern: |
          function $FUNC($ARGS) {
            ...
            $EXTERNAL_CALL(...);
            ...
            $STATE_MUTATION;
            ...
          }
    pattern-either:
      - pattern: $EXTERNAL_CALL(...);
      - pattern: $STATE_MUTATION;
    pattern-not: $SAFE_CALL(...);
    message: "State mutation following an external untrusted interaction detected."
    languages: [solidity]
    severity: WARNING
    metadata:
      author: "Your Name"
      description: "Detects state mutations following external calls to untrusted input."
```

```ql
import solidity

predicate isExternalCall(FunctionCall call) {
  call.getTarget().getName() = "external"
}

predicate isStateMutation(Assignment assign) {
  assign.getLeftHandSide() instanceof MemberAccess
}

from FunctionCall call, Assignment assign
where
  isExternalCall(call) and
  isStateMutation(assign) and
  call.getEnclosingCallable() = assign.getEnclosingCallable()
select assign, "State mutation following an external untrusted interaction detected."
```
